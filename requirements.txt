# Versi di bawah ini adalah yang benar-benar dipakai saat 51 test hijau,
# benchmark di docs/BENCHMARK.md dijalankan, dan endpoint /predict diuji.
# Lingkungan referensi: Windows 11 · Python 3.11.5 · CPU Intel (12th gen, iGPU).
#
# Pin ketat (==) supaya hasil orang lain sama persis dengan yang terdokumentasi.
# Kalau butuh lebih longgar (mis. Python 3.12+ yang belum punya wheel untuk
# salah satu versi ini), ganti `==` jadi `>=` per baris — bukan sekaligus —
# lalu jalankan `pytest tests -q` untuk memastikan tidak ada yang pecah.

# ---------------------------------------------------------------- inti model
ultralytics==8.4.117

# torch/torchvision ditarik otomatis oleh ultralytics, tapi di-pin eksplisit
# karena versinya menentukan angka di docs/BENCHMARK.md.
#
# PENTING untuk Linux/Docker: index PyPI default memberi wheel CUDA (~2.5 GB).
# Proyek ini hanya perlu CPU (OpenVINO yang menangani akselerasi). Install
# varian CPU lebih dulu supaya image tetap ramping:
#   pip install --index-url https://download.pytorch.org/whl/cpu \
#       torch==2.13.0 torchvision==0.28.0
# Di Windows/macOS wheel default sudah CPU-only, jadi tidak perlu langkah ini.
torch==2.13.0
torchvision==0.28.0

# ---------------------------------------------------------------- vision I/O
# PENTING: `roboflow` menarik opencv-python-headless, yang menimpa
# opencv-python sehingga cv2.imshow() mati (mode webcam CLI tidak jalan).
# Install opencv-python SETELAH roboflow, atau jalankan:
#   pip uninstall -y opencv-python-headless opencv-python
#   pip install opencv-python==4.10.0.84
opencv-python==4.10.0.84
numpy==2.3.5
Pillow==12.3.0

# ---------------------------------------------------------------- API & UI
fastapi==0.141.1
uvicorn[standard]==0.52.1
# Wajib untuk multipart upload di POST /predict — tanpa ini FastAPI menolak
# request dengan error 'python-multipart harus terinstall'.
python-multipart==0.0.32
# >=1.38: UI memakai st.audio(autoplay=True) untuk bunyi alarm.
streamlit==1.61.1

# ---------------------------------------------------------------- utilitas
python-dotenv==1.2.2
requests==2.34.2
# Backend `roboflow` (src/remote.py) — opsional, hanya untuk baseline online.
roboflow==1.4.0

# ---------------------------------------------------------------- OpenVINO
# Runtime backend OpenVINO (src/openvino_detector.py). Cukup untuk *menjalankan*
# IR yang sudah di-export; tanpa ini backend `openvino` / `openvino-int8` mati.
openvino==2026.3.0
# Dibutuhkan hanya saat *membuat* IR INT8 (scripts/export_openvino.py) —
# NNCF yang melakukan post-training quantization. Tidak dipakai saat inference,
# jadi aman dihapus di image deploy kalau mau menghemat ~100 MB.
nncf==3.3.0
# Export OpenVINO lewat Ultralytics melewati ONNX sebagai format perantara.
onnx==1.22.0

# Dibutuhkan HANYA saat export ONNX (scripts/export_fatigue.py). torch >= 2.9
# memakai exporter berbasis torch.export yang mengimpor onnxscript di level
# modul — jadi `import torch.onnx` gagal tanpa paket ini, bahkan ketika
# exporter lamanya (dynamo=False) yang dipakai. Aman dihapus di image deploy.
onnxscript==0.7.1

# ------------------------------------------------- fatigue detection & absensi
# Landmark wajah 478 titik + blendshape (src/fatigue/landmarks.py).
#
# PENTING: install dengan --no-deps. Dependency mediapipe menarik
# opencv-contrib-python 5.x yang akan MENIMPA opencv-python 4.10 di atas dan
# mengubah API yang dipakai src/detector.py. Yang benar-benar dibutuhkan cuma
# absl-py dan sounddevice, jadi keduanya dipasang terpisah:
#
#   pip install --no-deps mediapipe==1.0.1
#   pip install "absl-py>=2.0" "sounddevice~=0.5"
#
# Deteksi wajah (YuNet) dan pengenalan wajah (SFace) TIDAK butuh paket apa pun:
# keduanya jalan lewat cv2.FaceDetectorYN / cv2.FaceRecognizerSF yang sudah ada
# di opencv-python. Bobotnya diunduh otomatis oleh src/fatigue/assets.py.
mediapipe==1.0.1
absl-py==2.5.0
sounddevice==0.5.6

# scikit-learn: metrik evaluasi classifier fatigue (ROC-AUC, PR-AUC, confusion
# matrix) di scripts/train_fatigue.py & scripts/evaluate_fatigue.py.
# Tidak dipakai saat inference — aman dihapus di image deploy.
scikit-learn==1.9.0

# pandas: tabel di halaman Streamlit fatigue (app/fatigue_ui.py).
pandas==3.0.5
